// ============================================
// FIREBASE FIRESTORE COMPOSITE INDEXES
// ============================================
// Für die Firebase Console: Firestore Database > Indexes
// 
// Diese Indexes sind für die Party-Statistiken und Autostart-Funktionalität erforderlich
// ============================================

// ============================================
// INDEX 1: Wishes - Party-Statistiken (mit djId)
// ============================================
// Collection: wishes
// Fields:
//   - party_id (Ascending)
//   - djId (Ascending)
//   - status (Ascending)
// Query Scope: Collection
// 
// Verwendung: PartyStatisticsService.loadForPartyStream()
// Query: wishes.where('party_id', isEqualTo: partyId).where('djId', isEqualTo: djId)
// 
// JSON für Firebase Console:
{
  "collectionGroup": "wishes",
  "queryScope": "COLLECTION",
  "fields": [
    {
      "fieldPath": "party_id",
      "order": "ASCENDING"
    },
    {
      "fieldPath": "djId",
      "order": "ASCENDING"
    },
    {
      "fieldPath": "status",
      "order": "ASCENDING"
    }
  ]
}

      "order": "DESCENDING"
    }
  ]
}

// ============================================
// HINWEISE ZUR ERSTELLUNG:
// ============================================
// 1. Gehe zur Firebase Console: Firestore Database > Indexes
// 2. Klicke auf "Create Index"
// 3. Wähle die Collection Group aus
// 4. Füge die Felder in der angegebenen Reihenfolge hinzu
// 5. Wähle die Sortierrichtung (ASCENDING/DESCENDING)
// 6. Klicke auf "Create"
// 
// ALTERNATIV: Firebase erstellt die Indexes automatisch, wenn du die Queries ausführst
// und zeigt dir einen Link zur Index-Erstellung an.
// 
// WICHTIG: 
// - Indexes können einige Minuten zum Erstellen benötigen
// - Prüfe den Status in der Firebase Console
// - Queries funktionieren erst, wenn der Index erstellt wurde
// ============================================
